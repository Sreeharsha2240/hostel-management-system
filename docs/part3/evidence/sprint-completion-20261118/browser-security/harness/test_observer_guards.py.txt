class ResponseGuards(unittest.TestCase):
 def test_actual_200_dashboard_required(self):

   with self.subTest(status=status):self.assertFalse(protected_response_valid(SimpleNamespace(status=status,url='http://127.0.0.1:8109/dashboard/'),b'Warden workspace'))
  self.assertTrue(protected_response_valid(SimpleNamespace(status=200,url='http://127.0.0.1:8109/dashboard/'),b'Warden workspace'))
 def test_login_page_and_wrong_body_rejected(self):
  self.assertFalse(protected_response_valid(SimpleNamespace(status=200,url='http://127.0.0.1:8109/login/'),b'Warden workspace'))
  self.assertFalse(protected_response_valid(SimpleNamespace(status=200,url='http://127.0.0.1:8109/dashboard/'),b'Welcome to HMS'))

  self.assertEqual(result_exit_code('PASS'),0);self.assertEqual(result_exit_code('FAIL'),1);self.assertNotEqual(result_exit_code('PENDING'),0)
