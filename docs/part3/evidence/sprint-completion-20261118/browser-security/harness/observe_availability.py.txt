"""Real, fixed-window authenticated availability observation on the Linux host.

No clock changes, simulated probes, excluded failures or retrospective requests.
Resume preserves completed slots and records elapsed unobserved slots MISSING.
"""
import argparse
import fcntl
import hashlib
import http.cookiejar
import json
import math
import os
import re
import signal
import sys
import time
import urllib.parse
import urllib.request
from datetime import datetime, timedelta, timezone
from pathlib import Path
from zoneinfo import ZoneInfo

from matrix import source_snapshot

IST = ZoneInfo('Asia/Kolkata')
STOP = False


class DeadlineExceeded(Exception):
    pass


def stop_requested(signum, frame):
    global STOP
    STOP = True


def timeout_reached(signum, frame):
    raise DeadlineExceeded()


def write_json(path, value):
    temporary = path.with_suffix(path.suffix + '.new')
    temporary.write_text(json.dumps(value, indent=2))
    temporary.replace(path)


def validate_accepted_source(config, snapshot):
    expected=config.get('expected_source_pins')
    assert expected, 'Accepted source pins are required before observation'
    if isinstance(expected,str):expected=json.loads(Path(expected).read_text())
    expected=expected.get('pins',expected)
    assert expected and snapshot['pins']==expected, 'Runtime bytes differ from the accepted freeze'
    assert snapshot['commit']==config['expected_commit'], 'Wrong accepted source commit'
    return expected


def protected_response_valid(response, body):
    return response.status==200 and urllib.parse.urlparse(response.url).path=='/dashboard/' and b'warden workspace' in body.lower()


def result_exit_code(result):
    return 0 if result=='PASS' else 1 if result=='FAIL' else 2


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--config', type=Path, required=True)
    parser.add_argument('--output', type=Path, required=True)
    parser.add_argument('--date', default='2026-11-19')
    args = parser.parse_args()
    config = json.loads(args.config.read_text())
    assert config['authorized_frozen_run'] is True
    base = config['base_url'].rstrip('/')
    assert urllib.parse.urlparse(base).hostname in ['127.0.0.1', 'localhost']
    source = Path(config['source_root']).resolve()
    initial_source = source_snapshot(source)
    validate_accepted_source(config,initial_source)
    provenance_file=Path(config['server_provenance_file'])
    server_provenance=json.loads(provenance_file.read_text())
    assert server_provenance['source']['pins']==initial_source['pins'], 'Running server startup pins differ'
    assert server_provenance['source']['commit']==config['expected_commit']
    os.kill(server_provenance['pid'],0)
    assert b'serve.py' in Path('/proc/'+str(server_provenance['pid'])+'/cmdline').read_bytes(), 'Server process identity changed'
    start = datetime.fromisoformat(args.date + 'T08:00:00').replace(tzinfo=IST)
    end = datetime.fromisoformat(args.date + 'T20:00:00').replace(tzinfo=IST)
    assert start.weekday() < 5
    interval, timeout, total = 30, 5, 1440
    args.output.mkdir(parents=True, exist_ok=True)
    args.output.chmod(0o700)
    lock = (args.output / 'observer.lock').open('a+')
    fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
    header = {'case': 'TC-QUAL-02', 'window_start': start.isoformat(), 'window_end': end.isoformat(), 'timezone': 'Asia/Kolkata', 'interval_seconds': interval, 'timeout_seconds': timeout, 'planned_slots': total, 'minimum_successes': math.ceil(total * 0.99), 'endpoint': '/dashboard/', 'build': initial_source, 'policy': 'Missing and failed slots count against availability; each slot has a five-second deadline from its scheduled time; no exclusions or retrospective probe credit', 'limits': 'A twelve-hour local synthetic deployment observation with thirty-second sampling; not evidence for unobserved days or intervening outages', 'script_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(), 'source_helper_sha256':hashlib.sha256(Path(__file__).with_name('matrix.py').read_bytes()).hexdigest(), 'server_startup':server_provenance}
    header_file = args.output / 'observation-plan.json'
    if header_file.exists():
        assert json.loads(header_file.read_text()) == header
    else:
        write_json(header_file, header)
    records_file = args.output / 'probes.jsonl'
    rows = []
    if records_file.exists():
        rows = [json.loads(line) for line in records_file.read_text().splitlines() if line]
        assert [r['slot'] for r in rows] == list(range(len(rows)))
    jar = http.cookiejar.CookieJar()
    opener = urllib.request.build_opener(urllib.request.ProxyHandler({}), urllib.request.HTTPCookieProcessor(jar))
    checks_file = args.output / 'source-checks.jsonl'
    source_valid = not checks_file.exists() or all(json.loads(line)['source_unchanged'] for line in checks_file.read_text().splitlines() if line)
    auth = config['accounts']['warden']

    def login():
        url = base + '/login/'
        with opener.open(url, timeout=timeout) as response:
            assert response.status==200, 'Login form requires actual HTTP200'
            text = response.read().decode()
        token = re.search(r'name="csrfmiddlewaretoken" value="([^"]+)"', text).group(1)
        body = urllib.parse.urlencode({'username': auth['username'], 'password': auth['password'], 'csrfmiddlewaretoken': token}).encode()
        with opener.open(urllib.request.Request(url, data=body, headers={'Referer': url}), timeout=timeout) as response:
            assert protected_response_valid(response,response.read()), 'Authenticated login requires actual HTTP200 dashboard'

    def state(label):
        successes = sum(r['status'] == 'SUCCESS' for r in rows)
        failed = sum(r['status'] == 'FAIL' for r in rows)
        missing = sum(r['status'] == 'MISSING' for r in rows)
        complete = len(rows) == total and datetime.now(IST) >= end
        result = 'PENDING'
        if complete:
            result = 'PASS' if source_valid and successes >= math.ceil(total * .99) else 'FAIL'
        write_json(args.output / 'summary.json', {'case': 'TC-QUAL-02', 'state': label, 'result': result, 'updated_at': datetime.now(IST).isoformat(), 'window_start': start.isoformat(), 'window_end': end.isoformat(), 'recorded_slots': len(rows), 'planned_slots': total, 'successes': successes, 'failures': failed, 'missing': missing, 'success_percent_of_all_planned_slots': round(successes * 100 / total, 6), 'minimum_successes': math.ceil(total * .99), 'source_unchanged': source_valid, 'complete': complete, 'observer_pid': os.getpid()})
        return result

    signal.signal(signal.SIGINT, stop_requested)
    signal.signal(signal.SIGTERM, stop_requested)
    signal.signal(signal.SIGALRM, timeout_reached)
    state('WAITING' if datetime.now(IST) < start else 'IN_PROGRESS')
    print(json.dumps({'state': 'SCHEDULED', 'window_start': start.isoformat(), 'window_end': end.isoformat(), 'planned_slots': total, 'pid': os.getpid()}), flush=True)
    authenticated = False
    last_source_check = None
    with records_file.open('a', buffering=1) as log:
        for index in range(len(rows), total):
            planned = start + timedelta(seconds=index * interval)
            # Prepare a real login shortly before the first slot; it earns no probe credit.
            while datetime.now(IST) < planned and not STOP:
                remaining = (planned - datetime.now(IST)).total_seconds()
                if not authenticated and 0 < remaining <= 30:
                    try:
                        signal.setitimer(signal.ITIMER_REAL, timeout)
                        login()
                        authenticated = True
                    except Exception:
                        pass
                    finally:
                        signal.setitimer(signal.ITIMER_REAL, 0)
                    time.sleep(min(1,max(0,(planned-datetime.now(IST)).total_seconds())))
                else:
                    time.sleep(min(15, remaining))
            if STOP:
                state('INTERRUPTED')
                return 2
            observed = datetime.now(IST)
            row = {'slot': index, 'scheduled_at': planned.isoformat(), 'observed_at': observed.isoformat(), 'status': 'MISSING', 'reauthenticated': False}
            budget = timeout - (observed - planned).total_seconds()
            if budget > 0:
                tick = time.monotonic()
                try:
                    signal.setitimer(signal.ITIMER_REAL, budget)
                    with opener.open(base + '/dashboard/', timeout=budget) as response:
                        body = response.read()
                        row['http_status']=response.status
                        valid = protected_response_valid(response,body)
                    if not valid:
                        login()
                        row['reauthenticated'] = True
                        with opener.open(base + '/dashboard/', timeout=budget) as response:
                            body = response.read()
                            row['http_status']=response.status
                            valid = protected_response_valid(response,body)
                    if not valid:
                        raise ValueError('Unexpected protected response')
                    row.update({'status': 'SUCCESS', 'body_sha256': hashlib.sha256(body).hexdigest()})
                    authenticated = True
                except Exception as error:
                    row.update({'status': 'FAIL', 'error_type': type(error).__name__})
                finally:
                    signal.setitimer(signal.ITIMER_REAL, 0)
                    row['elapsed_ms'] = round((time.monotonic() - tick) * 1000, 3)
            else:
                row['reason'] = 'Scheduled five-second observation opportunity elapsed without a retained probe'
            row['completed_at'] = datetime.now(IST).isoformat()
            log.write(json.dumps(row) + '\n')
            log.flush()
            os.fsync(log.fileno())
            rows.append(row)
            if last_source_check is None or index % 120 == 0:
                current = source_snapshot(source)
                unchanged = current['pins'] == initial_source['pins']
                source_valid = source_valid and unchanged
                with checks_file.open('a') as checks:
                    checks.write(json.dumps({'at': datetime.now(IST).isoformat(), 'source_unchanged': unchanged, 'commit': current['commit']}) + '\n')
                last_source_check = index
            state('IN_PROGRESS')
        while datetime.now(IST) < end and not STOP:
            time.sleep(min(15, (end - datetime.now(IST)).total_seconds()))
    final_source = source_snapshot(source)
    source_valid = source_valid and final_source['pins'] == initial_source['pins']
    write_json(args.output / 'source-final.json', final_source)
    result=state('COMPLETE' if not STOP else 'INTERRUPTED')
    return result_exit_code(result)


if __name__ == '__main__':
    raise SystemExit(main())
